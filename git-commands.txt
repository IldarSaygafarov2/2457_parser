git -v проверка версии гита на компе

.gitignore - файл, с названиями папок и файлов, которые не должны отправляться в репозиторий
git init - инициализация пустого репозитория в проекте
git add . - добавление всех файлов всех папок не добавленых в .gitignore

git config user.name "username" - добавляем имя пользователя отправителя
git config user.email "email" - добавляем почту отправителя

(прописывается 1 раз)
git config --global user.name "username" - глобально добавляем имя пользователя отправителя
git config --global user.email "email" - глобально добавляем почту отправителя

git commit -m "message" - добавление комментария к отправляемым файлам

(прописывается 1 раз)
git branch -M main - создание основной ветки main
git remote add origin ссылка_с_репозитория - связывает локальный репозиторий с репозиторием на гитхаб
git push -u origin main - первая отправка данных в репозиторий


(отправка данных после внесения изменений)
git add .
git commit -m "message"
git push
